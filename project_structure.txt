FLUJO DE EJECUCIÓN
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

1. INSTALAR DEPENDENCIAS
   pip install -r requirements.txt

2. ENTRENAR EL MODELO
   python train.py --dataset dataset/ --epochs 40 --device auto

3. (OPCIONAL) VERIFICAR COORDENADAS
   python calibrate.py --pdf plantilla.pdf --page 2

4. DETECTAR ANOMALÍAS
   python infer.py --pdfs carpeta_pdfs/ --threshold 0.85

5. REVISAR RESULTADOS
   output/reports/reporte_*.xlsx

PÁGINAS DEL FORMULARIO (30 páginas)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

Pág  1        nivelacion          Nivelación de la mesa (3 campos × 3 dígitos)
Págs 2-3      preferente_full     Circ. Nacional — lista voto preferente (100 cand.)
Págs 4-5      sin_preferente      Circ. Nacional — lista sin voto preferente
Pág  6        preferente_full     Circ. Nacional — lista voto preferente (100 cand.)
Págs 7-9      sin_preferente      Circ. Nacional — listas sin voto preferente
Págs 10-13    preferente_full     Alianzas para curules — voto preferente
Págs 14-17    preferente_full     Alianzas para curules — voto preferente
Pág  18       totales_nacionales  Totales Circ. Nacional (blanco, nulos, no marcados)
Págs 19-21    sin_preferente      Circ. Especial Comunidades Indígenas
Pág  22       preferente_full     Circ. Especial — Movimiento SI
Págs 23-30    skip                Certificaciones, observaciones, firmas (sin dígitos)

CONDICIONES DE ANOMALÍA (por slot de dígito)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  baja_confianza        softmax_max < 0.85
  entropia_alta         H(softmax) > 1.20
  margen_insuficiente   top1 − top2 < 0.25
  multiples_componentes n_componentes_CC > 1
  caracter_fusionado    bbox_ratio > 0.95
  fragmento_pequeno     bbox_ratio < 0.08
